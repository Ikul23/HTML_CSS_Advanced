# Структура проекта под SPA React

│
├── public/ # Папка для статических файлов (HTML, изображения, шрифты)
│ ├── assets/
│ │ ├── images/ # Изображения
│ │ └── fonts/ # Шрифты
│ └── index.html # Главный HTML-файл
│
├── src/ # Исходные файлы (SCSS, JS, компоненты)
│ ├── styles/ # Стили проекта
│ │ ├── \_vars.scss # Переменные SASS (цвета, шрифты, размеры)
│ │ ├── main.scss # Главный файл SCSS (импортирует все остальные)
│ │ ├── main.css # Скомпилированный CSS (будет создан автоматически)
│ │ ├── global.scss # Глобальные стили (сброс, шрифты, переменные)
│ │ ├── components/ # Стили для компонентов
│ │ │ ├── \_header.scss # Стили для Header
│ │ │ ├── \_footer.scss # Стили для Footer
│ │ │ └── \_buttons.scss # Стили для кнопок
│ │ └── pages/ # Стили для страниц
│ │ ├── \_index.scss # Стили для главной страницы
│ │ ├── \_catalog.scss # Стили для каталога
│ │ ├── \_product.scss # Стили для страницы продукта
│ │ ├── \_cart.scss # Стили для корзины
│ │ └── \_registration.scss # Стили для регистрации
│ │
│ └── scripts/ # Скрипты (если есть)
│ └── main.js # Главный JS-файл
│
└── package.json # Файл с зависимостями и скриптами (если используется npm)
